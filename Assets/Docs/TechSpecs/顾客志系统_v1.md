```csharp
public class CustomerRecord{
    public MetaCustomer metaCustomer;
    public int visitCount;
    public int totalSpent;
    
    // === 新增字段 ===
    public int lastVisitDay;  // 上次来访的日期（GameSystem.CurrentDay）

        this.totalSpent = 0;
        this.lastVisitDay = 0;  // 初始化为0表示未来过
    }

```

**CustomerRecorder.OnServeCustomer() 方法修改：**

```csharp
public void OnServeCustomer(DealContext context, DealResult result){
    // 如果已经在顾客志中
    if (!customerRecords.Any(record => record.metaCustomer == context.Customer.meta)){
        AddRecord(context.Customer.meta);
    }

    CustomerRecord record = customerRecords.FirstOrDefault(record => record.metaCustomer == context.Customer.meta);
    record.visitCount++;
    record.totalSpent += result.price;
    
    // === 新增：记录上次来访日期 ===
    record.lastVisitDay = this.GetSystem<IGameSystem>().CurrentDay;
}
```

```csharp
using QFramework;
using Sirenix.OdinInspector;
using TMPro;
using UniRx;
using UnityEngine;

public class TopBlankController : MonoBehaviour, IController
{
    [SerializeField] private TextMeshProUGUI coinText;
    [SerializeField] private TextMeshProUGUI reputationText;
    [SerializeField] private TextMeshProUGUI levelText;
    [SerializeField] private TextMeshProUGUI incomeText;
    
    // === 新增：时段显示 ===
    [LabelText("时段显示")]
    [SerializeField] private TextMeshProUGUI customerTimeText;
    
    public IArchitecture GetArchitecture() => GameArchitecture.Interface;

    void Start()
    {
        // ... 已有订阅代码 ...
        
        // === 新增：订阅时段变化（通过日开始事件刷新） ===
        this.RegisterEvent<StartNewDayEvent>(OnStartNewDay).AddTo(this);
        
        // 初始化时段显示
        UpdateCustomerTimeInfo();
    }
    
    private void OnStartNewDay(StartNewDayEvent evt){
        if (evt.StageMeet(EventStage.After)){
            UpdateCustomerTimeInfo();
        }
    }
    
    // === 新增：更新时段显示 ===
    private void UpdateCustomerTimeInfo(){
        if (customerTimeText != null){
            string displayName = this.GetSystem<ICustomerSystem>().GetCustomerTimeDisplayName();
            float chance = this.GetSystem<ICustomerSystem>().GetOldCustomerChance();
            customerTimeText.text = $"{displayName} 老客:{(int)(chance * 100)}%";
        }
    }
    
    // ... 已有方法 ...
}
```

### 2.4 Tooltip 配置

#### 方式一：使用 StaticTooltipShower（推荐用于静态文本）

1. **创建 TooltipInfoSO 资源文件：**
   - 路径：`Assets/Resources/Tooltip/CustomerTime_Morning.asset`（午市）
   - 路径：`Assets/Resources/Tooltip/CustomerTime_Night.asset`（夜市）

2. **TooltipInfoSO 配置示例（午市）：**
```
tooltipInfos:
  - text: "☀️ 午市"
    style: Title
  - text: "────────────────"
    style: Normal
  - text: "老客概率: 50%"
    style: Highlight
  - text: "新客概率: 50%"
    style: Normal
  - text: "────────────────"
    style: Normal
  - text: "适合积累熟客关系"
    style: Description
  - text: "更容易遇到老顾客"
    style: Description
```

#### 方式二：动态 Tooltip（推荐用于动态内容）

如果需要根据当前时段动态生成 Tooltip，可以创建新的 Shower 组件：

```csharp
// 文件路径：Assets/Scripts/UI/Tooltip/CustomerTimeTooltipShower.cs
using System.Collections.Generic;
using Sirenix.OdinInspector;
using UnityEngine;

/// <summary>
/// 顾客时段动态Tooltip显示器
/// </summary>
public class CustomerTimeTooltipShower : SerializedMonoBehaviour, IShowTooltip
{
    public List<TooltipInfo> GetTooltipInfo(){
        var customerSystem = GameArchitecture.Interface.GetSystem<ICustomerSystem>();
        string displayName = customerSystem.GetCustomerTimeDisplayName();
        float oldChance = customerSystem.GetOldCustomerChance();
        float newChance = 1f - oldChance;
        
        return new List<TooltipInfo>{
            new TooltipInfo(displayName, TooltipStyle.Title),
            new TooltipInfo("────────────────", TooltipStyle.Normal),
            new TooltipInfo($"老客概率: {(int)(oldChance * 100)}%", TooltipStyle.Highlight),
            new TooltipInfo($"新客概率: {(int)(newChance * 100)}%", TooltipStyle.Normal),
            new TooltipInfo("────────────────", TooltipStyle.Normal),
            new TooltipInfo(customerSystem.CustomerTime == CustomerTime.午间 ? 
                "适合积累熟客关系" : "人流量大更热闹", TooltipStyle.Description),
        };
    }
}
```

## 四、注意事项

1. **遵循 QFramework 规范**：System 层不依赖 UnityEngine 的 Transform、GameObject 等
2. **使用 UniRx**：事件订阅使用 `.AddTo(this)` 确保生命周期管理
3. **命名规范**：UI 组件以类型结尾（如 `customerTimeText`）
4. **注释规范**：新增方法必须包含 `<summary>` 注释
